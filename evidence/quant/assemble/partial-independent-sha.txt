## tensor_sha.py
# tensor_sha.py FILE... : prints "sha256 dtype shape name" per tensor, sorted by name. With EXPERTS=skip, routed expert .weight is skipped;
# with EXPERTS=only, only routed expert parts are printed.
import hashlib, json, os, re, struct, sys
mode = os.environ.get("EXPERTS", "")
pat = re.compile(r"model\.language_model\.layers\.\d+\.mlp\.experts\.\d+\.(gate|up|down)_proj\.")
rows = []
for p in sys.argv[1:]:
    with open(p, "rb") as f:
        n = struct.unpack("<Q", f.read(8))[0]; h = json.loads(f.read(n)); h.pop("__metadata__", None)
        for k, v in h.items():
            e = bool(pat.match(k))
            if (mode == "skip" and e) or (mode == "only" and not e): continue
            a, b = v["data_offsets"]; f.seek(8 + n + a)
            rows.append(f"{hashlib.sha256(f.read(b - a)).hexdigest()} {v['dtype']} {v['shape']} {k}".replace(", ", ","))
print("\n".join(sorted(rows, key=lambda r: r.rsplit(" ", 1)[1])))
